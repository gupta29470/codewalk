# ── A1. LLM (required — cloud server has no Ollama) ───────────────────
# Where: pick a cloud provider and get API key from their dashboard
#
#   groq      → https://console.groq.com/keys
#   deepseek  → https://platform.deepseek.com/api_keys
#   openai    → https://platform.openai.com/api-keys
#   anthropic → https://console.anthropic.com/settings/keys
#   openrouter→ https://openrouter.ai/keys
#
LLM_PROVIDER=deepseek
LLM_MODEL=deepseek-chat

# Fill in ONLY the key for your chosen provider:
GROQ_API_KEY=
OPENAI_API_KEY=
ANTHROPIC_API_KEY=
GOOGLE_API_KEY=
OPENROUTER_API_KEY=
DEEPSEEK_API_KEY=

# ── A2. Embeddings (required) ─────────────────────────────────────────
# Where: HuggingFace model name — downloaded automatically on first index
# Default works for most setups. First index downloads ~3GB model.
EMBEDDING_MODEL=jinaai/jina-code-embeddings-1.5b

# ── A3. Database (required on server) ─────────────────────────────────
# Where: YOU choose a strong password. Generate with:
#   openssl rand -base64 24
# IMPORTANT: No @ : / # characters — they break DATABASE_URL in docker-compose.
# If you change password after first deploy, run ALTER USER in postgres (see FULL_SETUP_GUIDE.md).
#
# docker-compose.yml auto-builds DATABASE_URL from this — you do NOT
# need a separate DATABASE_URL line if using deploy/docker-compose.yml.
#
POSTGRES_PASSWORD=

# Only needed if your compose uses DATABASE_URL=${DATABASE_URL:-}
# (hetzner-setup.sh old template). Main deploy/docker-compose.yml does NOT need this.
# DATABASE_URL=postgresql://codewalk:YOUR_POSTGRES_PASSWORD@postgres/codewalk

# ── A4. Cloud indexing (required for GitHub webhook auto-index) ───────
#
# GITHUB_APP_ID
#   Where: GitHub → Settings → Developer settings → GitHub Apps
#          → your "Codewalk Cloud" app → App ID at top of page
#
# GITHUB_APP_PRIVATE_KEY_PATH
#   Where: Path to downloaded .pem inside container, e.g.
#          /var/codewalk/secrets/codewalk-cloud.private-key.pem
#          (GitHub App → Private keys → Generate → download .pem)
#
# GITHUB_WEBHOOK_SECRET
#   Where: Same app page → Webhook → Webhook secret
#          Generate with: openssl rand -hex 32
#          Must match the secret set when creating the GitHub App
#
# ADMIN_API_KEY
#   Where: YOU generate — any random string. Used for /admin UI + API.
#          Generate with: openssl rand -hex 32
#
GITHUB_APP_ID=
GITHUB_APP_PRIVATE_KEY_PATH=/var/codewalk/secrets/codewalk-cloud.private-key.pem
GITHUB_WEBHOOK_SECRET=
ADMIN_API_KEY=

# ── A5. Server tuning (optional) ──────────────────────────────────────
RATE_LIMIT_REQUESTS=60
RATE_LIMIT_WINDOW=60
CORS_ORIGINS=*
INDEX_STORAGE_PATH=/var/codewalk
# Server only: reset repos indexing longer than N minutes (watchdog + catch-up)
CODEWALK_STUCK_INDEX_MINUTES=30

# ═══════════════════════════════════════════════════════════════════════
#  SECTION C — MCP per-project (.vscode/mcp.json env block)
#  Used by: VS Code / Cursor Copilot when analyzing a target repo
# ═══════════════════════════════════════════════════════════════════════
#
#  {
#    "servers": {
#      "codewalk": {
#        "command": "/path/to/codewalk/.codewalk-env/bin/python",
#        "args": ["-m", "src.codewalk.mcp.server"],
#        "cwd": "${workspaceFolder}",
#        "env": {
#          "--- Cloud MCP (download index from server): ---"
#          "CODEWALK_SERVER_URL": "https://api.codewalk.xyz",
#          "CODEWALK_REPO_NAME": "your-org/flutter-gallery",
#          "CODEWALK_REPO_TOKEN": "cw_repo_xxxxxxxx"
#        }
#      }
#    }
#  }
#
#  Where to get each:
#    cwd                  → ${workspaceFolder} — MUST be the same repo as CODEWALK_REPO_NAME
#                           (open that repo in Cursor/VS Code; index lands in {cwd}/.codewalk/)
#    CODEWALK_SERVER_URL  → your Hetzner domain or IP (Caddy :443)
#    CODEWALK_REPO_NAME   → owner/repo slug from GitHub
#    CODEWALK_REPO_TOKEN  → on server after repo registers:
#                           docker compose exec postgres psql -U codewalk -d codewalk -c \
#                             "SELECT repo_token FROM repos WHERE full_name='owner/repo';"
#                           (admin/repos does not return token yet)
#
#  Review/HITL: IDE agent calls MCP tools; each host has its own approve/reject UI.
#    Per fix: codewalk_approve_action → user approves in host UI → codewalk_apply_fix(..., approval_token=...).
#    Rules: src/codewalk/mcp/server.py FastMCP instructions.
